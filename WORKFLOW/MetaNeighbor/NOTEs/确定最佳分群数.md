# 答
我来帮你添加多种方法来确定最佳分群数 k。常用方法包括**轮廓系数（Silhouette）**、**肘部法则（Elbow）** 和 **间隙统计（Gap Statistic）**：

---

## 完整代码：确定最佳分群数

```r
library(gplots)
library(factoextra)    # 可视化聚类
library(NbClust)       # 多种指标确定最佳k
library(cluster)       # 轮廓系数

# ========== 1. 计算距离矩阵和聚类 ==========
dist_matrix <- dist(celltype_NV)
row_hclust <- hclust(dist_matrix)

# ========== 方法1: 轮廓系数法（推荐）==========
# 计算不同k的轮廓系数，选择平均值最高的k
silhouette_scores <- sapply(2:10, function(k) {
  groups <- cutree(row_hclust, k = k)
  sil <- silhouette(groups, dist_matrix)
  mean(sil[, 3])  # 第3列是轮廓宽度
})

best_k_sil <- which.max(silhouette_scores) + 1  # +1因为k从2开始

cat("轮廓系数法最佳k:", best_k_sil, "\n")
cat("各k的轮廓系数:", round(silhouette_scores, 3), "\n")

# 可视化轮廓系数
plot(2:10, silhouette_scores, type = "b", pch = 19, col = "blue",
     xlab = "Number of clusters (k)", ylab = "Average Silhouette Width",
     main = "Silhouette Method for Optimal k")
abline(v = best_k_sil, col = "red", lty = 2)

# ========== 方法2: 肘部法则（Within-cluster Sum of Squares）==========
wss <- sapply(1:10, function(k) {
  groups <- cutree(row_hclust, k = k)
  # 计算总within-cluster sum of squares
  total_wss <- sum(by(dist_matrix, groups, function(x) sum(as.matrix(x)^2)))
  total_wss
})

# 寻找肘部（二阶导数最大点）
elbow_k <- which.max(diff(diff(wss))) + 1

cat("\n肘部法则建议k:", elbow_k, "\n")

plot(1:10, wss, type = "b", pch = 19, col = "darkgreen",
     xlab = "Number of clusters (k)", ylab = "Within-cluster Sum of Squares",
     main = "Elbow Method")
abline(v = elbow_k, col = "red", lty = 2)

# ========== 方法3: 间隙统计（Gap Statistic）==========
set.seed(123)
gap_stat <- clusGap(celltype_NV, FUN = hcut, K.max = 10, B = 50)
best_k_gap <- maxSE(f = gap_stat$Tab[, "gap"], SE.f = gap_stat$Tab[, "SE.f"])

cat("\n间隙统计法最佳k:", best_k_gap, "\n")
fviz_gap_stat(gap_stat)

# ========== 方法4: NbClust综合多种指标（最全面）==========
# 注意：计算较慢，测试多种指标
nb_results <- NbClust(celltype_NV, distance = "euclidean", 
                      min.nc = 2, max.nc = 10, 
                      method = "ward.D2", index = "all")

cat("\nNbClust多数投票最佳k:", nb_results$Best.nc[1], "\n")
cat("各指标建议的k值:\n")
print(table(nb_results$Best.nc[1,]))

# ========== 综合决策 ==========
cat("\n========== 综合结果 ==========\n")
cat("轮廓系数法:", best_k_sil, "\n")
cat("肘部法则:", elbow_k, "\n")
cat("间隙统计法:", best_k_gap, "\n")
cat("NbClust投票:", as.numeric(names(which.max(table(nb_results$Best.nc[1,])))), "\n")

# 选择最常见的k（或根据业务选择）
best_k <- as.numeric(names(which.max(table(c(best_k_sil, elbow_k, best_k_gap, 
                                               nb_results$Best.nc[1,])))))
cat("\n★ 推荐最佳k:", best_k, "\n")

# ========== 使用最佳k进行分群 ==========
row_groups <- cutree(row_hclust, k = best_k)

cat("\n各群样本数:\n")
print(table(row_groups))

# ========== 可视化最终聚类结果 ==========
# 绘制带分群边界的树状图
plot(row_hclust, main = paste("Hierarchical Clustering (k =", best_k, ")"))
rect.hclust(row_hclust, k = best_k, border = 2:(best_k+1))

# 绘制热图
heatmap.2(celltype_NV,
          col = colorRampPalette(c("blue", "white", "red"))(100),
          Rowv = as.dendrogram(row_hclust),
          Colv = TRUE,
          key.xlab = "AUROC",
          margins = c(8, 8),
          trace = "none",
          density.info = "none",
          cexRow = 0.7,
          cexCol = 0.7,
          main = paste("Heatmap with k =", best_k, "clusters"))

# 保存分群结果
cluster_result <- data.frame(
  sample = rownames(celltype_NV),
  cluster = row_groups
)
write.csv(cluster_result, "optimal_clusters.csv", row.names = FALSE)
```

## 各方法原理速查

| 方法 | 原理 | 优点 | 缺点 |
|-----|------|------|------|
| **轮廓系数** | 衡量样本与自身群 cohesion vs 与其他群 separation | 直观，-1到1范围 | 对凸形群效果差 |
| **肘部法则** | WSS随k增加而下降，找下降速率突变点 | 简单直观 | 主观判断"肘部"位置 |
| **间隙统计** | 比较实际WSS与随机数据的WSS差距 | 有统计基础，自动确定 | 计算量大 |
| **NbClust** | 综合30种指标投票 | 最全面，减少单一指标偏差 | 计算慢，可能不一致 |

## 快速版（仅轮廓系数）

如果只需要快速结果：

```r
library(gplots)
library(cluster)

# 快速确定最佳k
row_hclust <- hclust(dist(celltype_NV))

sil_scores <- sapply(2:8, function(k) {
  mean(silhouette(cutree(row_hclust, k), dist(celltype_NV))[, 3])
})

best_k <- which.max(sil_scores) + 1
cat("最佳k =", best_k, "\n")

# 使用最佳k
row_groups <- cutree(row_hclust, k = best_k)
```
